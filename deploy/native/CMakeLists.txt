cmake_minimum_required(VERSION 3.24)
project(NeuralFXNative VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORT_ROOT "" CACHE PATH "Unpacked official ONNX Runtime CPU SDK")
set(JUCE_ROOT "" CACHE PATH "Optional JUCE source tree")
if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set ORT_ROOT to an unpacked ONNX Runtime SDK")
endif()
if(WIN32)
  # Windows ships a different onnxruntime.dll. A private import name prevents DAW/system collisions.
  set(NFX_ORT_RUNTIME_FILENAME "neural_fx_ort_123.dll")
  set(ORT_LIBRARY "")
else()
  find_library(ORT_LIBRARY NAMES onnxruntime PATHS "${ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
endif()
add_library(neural_fx_core STATIC processor.cpp)
target_include_directories(neural_fx_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${ORT_ROOT}/include")
target_link_libraries(neural_fx_core PUBLIC "${ORT_LIBRARY}")
target_compile_definitions(neural_fx_core PUBLIC ORT_API_MANUAL_INIT=1)
set_target_properties(neural_fx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(neural_fx_core PUBLIC /EHsc PRIVATE /W4)
endif()
add_executable(nafx-render main.cpp allocation_probe.cpp)
target_link_libraries(nafx-render PRIVATE neural_fx_core)
if(WIN32)
  add_custom_command(TARGET nafx-render POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:nafx-render>/${NFX_ORT_RUNTIME_FILENAME}")
endif()
if(JUCE_ROOT)
  add_subdirectory("${JUCE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
  add_subdirectory(../juce "${CMAKE_CURRENT_BINARY_DIR}/plugin")
endif()
